import React, { useState } from "react";
import { View, Text, ScrollView, TouchableOpacity, ActivityIndicator } from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { Feather } from "@expo/vector-icons";
import { useRouter } from "expo-router";
import { useTheme } from "@/hooks/useTheme";
import { Button } from "@/components/ui/Button";
import { BlurView } from "expo-blur";
import { useAuthStore } from "@/store/auth";
import * as WebBrowser from "expo-web-browser";
import { useMutation } from "@apollo/client";
import { REFRESH_NFT_STATUS } from "@/services/queries";

export default function NFTPurchaseScreen() {
  const { colors } = useTheme();
  const router = useRouter();
  const [isVerifying, setIsVerifying] = useState(false);
  const [verificationError, setVerificationError] = useState<string | null>(null);
  
  const user = useAuthStore((state) => state.user);
  const hasNFTAccess = useAuthStore((state) => state.hasNFTAccess);
  const setNFTStatus = useAuthStore((state) => state.setNFTStatus);
  
  // GraphQL Mutation for refreshing NFT status
  const [refreshNFTStatus] = useMutation(REFRESH_NFT_STATUS, {
    onCompleted: async (data) => {
      const status = data.refreshNFTStatus;
      if (status?.hasAccess) {
        const tokens =
          status.tokens?.map((token: { tokenMint: string }) => token.tokenMint) ??
          [];
        await setNFTStatus(true, tokens);
      } else {
        setVerificationError("No NFT found in your wallet");
      }
    },
    onError: (error) => {
      console.error("Error refreshing NFT status:", error);
      setVerificationError("Failed to verify NFT ownership");
    },
  });

  const verifyNFT = async (retryCount = 0) => {
    if (!user) return;
    
    const MAX_RETRIES = 2;
    const RETRY_DELAY = 2000; // 2 seconds
    
    try {
      setIsVerifying(true);
      setVerificationError(null);
      
      // In devnet, simulate verification
      if (process.env.EXPO_PUBLIC_NETWORK === "devnet") {
        console.log("Running in devnet - simulating NFT verification");
        // Simulate network delay
        await new Promise<void>((resolve) => setTimeout(resolve, 1000));
        await setNFTStatus(true);
        return true;
      }
      
      // In production, call the GraphQL mutation
      const { data } = await refreshNFTStatus();

      if (!data?.refreshNFTStatus?.hasAccess) {
        throw new Error("NFT verification failed");
      }

      const tokens =
        data.refreshNFTStatus.tokens?.map(
          (token: { tokenMint: string }) => token.tokenMint
        ) ?? [];

      await setNFTStatus(true, tokens);
      return true;
      
    } catch (error: any) {
      console.error('NFT verification failed:', error);
      
      // Implement retry logic
      if (retryCount < MAX_RETRIES) {
        setVerificationError('Retrying verification...');
        await new Promise<void>(resolve => setTimeout(resolve, RETRY_DELAY));
        return verifyNFT(retryCount + 1);
      }
      
      setVerificationError(
        error.message.includes('network')
          ? 'Network error. Please check your connection.'
          : 'Failed to verify NFT ownership. Please try again.'
      );
      return false;
      
    } finally {
      setIsVerifying(false);
    }
  };

  const handlePurchaseNFT = async () => {
    try {
    const marketplaceUrl =
      process.env.EXPO_PUBLIC_NETWORK === "mainnet-beta"
        ? "https://magiceden.io/marketplace/gocabs"
        : "https://solanadevnet.gatsbyjs.io/";
      
      const result = await WebBrowser.openBrowserAsync(marketplaceUrl);
      
      if (result.type === 'dismiss') {
        const success = await verifyNFT();
        if (success) {
          // Show success state briefly before navigating
          await new Promise<void>(resolve => setTimeout(resolve, 1000));
          router.replace('/(tabs)');
        }
      }
    } catch (error) {
      console.error('Error in purchase flow:', error);
      setVerificationError('Failed to complete purchase. Please try again.');
    }
  };

  const handleBack = () => {
    router.back();
  };

  return (
    <SafeAreaView className="flex-1 bg-background" edges={["top", "bottom"]}>
      {/* Header */}
      <View className="flex-row items-center px-4 py-4">
        <TouchableOpacity
          onPress={handleBack}
          className="w-10 h-10 rounded-full bg-background/30 items-center justify-center"
        >
          <BlurView
            intensity={20}
            className="w-full h-full rounded-full items-center justify-center"
          >
            <Feather name="arrow-left" size={20} color="#fff" />
          </BlurView>
        </TouchableOpacity>
        <Text className="text-white text-xl font-heading-bold ml-4">
          Get GoCabs NFT
        </Text>
      </View>

      <ScrollView className="flex-1" showsVerticalScrollIndicator={false}>
        <View className="px-4 py-6">
          {/* Hero Section */}
          <View className="items-center mb-8">
            <View className="w-32 h-32 rounded-3xl bg-primary/20 items-center justify-center mb-6 border border-primary/40">
              <Feather name="image" size={64} color={colors.primary} />
            </View>
            <Text className="text-white text-3xl font-heading-bold text-center mb-3">
              Unlock Full Access
            </Text>
            <Text className="text-white/70 text-center font-body text-base px-6">
              Own a GoCabs NFT to access all premium features
            </Text>
          </View>

          {/* Benefits */}
          <View className="mb-8">
            <Text className="text-white text-xl font-heading-bold mb-4">
              NFT Holder Benefits
            </Text>

            {[
              {
                icon: "calendar",
                title: "Book Rides",
                description: "Reserve shuttle services for events",
              },
              {
                icon: "map-pin",
                title: "Live Tracking",
                description: "Track your shuttle in real-time",
              },
              {
                icon: "award",
                title: "Earn Rewards",
                description: "Collect XP and redeem for USDC",
              },
              {
                icon: "trending-up",
                title: "Revenue Share",
                description: "Stake NFTs to earn platform revenue",
              },
              {
                icon: "percent",
                title: "Fractional Ownership",
                description: "Link NFTs to shuttles for revenue",
              },
            ].map((benefit, index) => (
              <View
                key={index}
                className="bg-card/80 rounded-2xl p-4 mb-3 border border-border/20"
              >
                <View className="flex-row items-start">
                  <View className="w-10 h-10 bg-primary/20 rounded-full items-center justify-center mr-3">
                    <Feather name={benefit.icon as any} size={20} color={colors.primary} />
                  </View>
                  <View className="flex-1">
                    <Text className="text-white font-body-bold text-base mb-1">
                      {benefit.title}
                    </Text>
                    <Text className="text-white/60 font-body text-sm">
                      {benefit.description}
                    </Text>
                  </View>
                </View>
              </View>
            ))}
          </View>

          {/* Pricing Info */}
          <View className="bg-primary/10 rounded-2xl p-6 mb-6 border border-primary/30">
            <View className="flex-row items-center justify-between mb-4">
              <Text className="text-white font-body-bold text-lg">
                Current Price
              </Text>
              <Text className="text-primary font-heading-bold text-2xl">
                TBD SOL
              </Text>
            </View>
            <Text className="text-white/60 font-body text-sm">
              Price subject to change based on market conditions. Visit Magic Eden
              for current pricing and availability.
            </Text>
          </View>

          {/* How to Purchase */}
          <View className="mb-8">
            <Text className="text-white text-xl font-heading-bold mb-4">
              How to Purchase
            </Text>

            {[
              {
                step: "1",
                title: "Connect Wallet",
                description: "Use Phantom, Solflare, or another Solana wallet",
              },
              {
                step: "2",
                title: "Visit Marketplace",
                description: "Go to Magic Eden or our official marketplace",
              },
              {
                step: "3",
                title: "Purchase NFT",
                description: "Complete your purchase with SOL",
              },
              {
                step: "4",
                title: "Refresh Status",
                description: "Return to the app and refresh your NFT status",
              },
            ].map((item, index) => (
              <View key={index} className="flex-row mb-4">
                <View className="w-8 h-8 rounded-full bg-primary items-center justify-center mr-4">
                  <Text className="text-black font-body-bold text-sm">
                    {item.step}
                  </Text>
                </View>
                <View className="flex-1">
                  <Text className="text-white font-body-bold text-base mb-1">
                    {item.title}
                  </Text>
                  <Text className="text-white/60 font-body text-sm">
                    {item.description}
                  </Text>
                </View>
              </View>
            ))}
          </View>

          <View className="h-24" />
        </View>
      </ScrollView>

      {/* Bottom CTA */}
      <View className="px-4 pb-6 pt-4 bg-black/50">
        <BlurView intensity={80} className="absolute inset-0" tint="dark" />
        
        {isVerifying ? (
          <View className="bg-primary/20 p-4 rounded-xl items-center">
            <ActivityIndicator size="small" color={colors.primary} className="mb-2" />
            <Text className="text-white font-body text-center">
              Verifying NFT ownership...
            </Text>
          </View>
        ) : hasNFTAccess ? (
          <View className="bg-green-500/20 p-4 rounded-xl items-center">
            <Feather name="check-circle" size={24} color="#10B981" className="mb-2" />
            <Text className="text-green-400 font-body-bold text-center text-base">
              NFT Verified! You have full access.
            </Text>
            <Text className="text-green-300/80 font-body text-center text-sm mt-1">
              Enjoy all premium features.
            </Text>
          </View>
        ) : (
          <View>
            <Button
              title={process.env.EXPO_PUBLIC_NETWORK === 'devnet' 
                ? "Simulate NFT Purchase (Devnet)" 
                : "Purchase on Magic Eden"}
              onPress={handlePurchaseNFT}
              variant="primary"
              size="large"
              icon="external-link"
              iconPosition="right"
              disabled={isVerifying}
            />
            
            {verificationError && (
              <Text className="text-red-400 text-center text-sm mt-2">
                {verificationError}
              </Text>
            )}
            
            <TouchableOpacity
              onPress={() => verifyNFT()}
              className="mt-3"
              disabled={isVerifying}
            >
              <Text className="text-white/60 text-center font-body text-sm">
                I already own an NFT - Verify Ownership
              </Text>
            </TouchableOpacity>
          </View>
        )}
      </View>
    </SafeAreaView>
  );
}



import { create } from "zustand";
import * as SecureStore from "expo-secure-store";
import { updateAuthToken } from "../services/apollo";
import { apolloClient } from "@/services/apollo";
import { GET_ME } from "@/services/queries";
import { User } from "@/types/user";

interface AuthState {
  user: User | null;
  token: string | null;
  hasNFTAccess: boolean;
  nftTokens: string[];
  isAuthenticated: boolean;
  isLoading: boolean;
  setAuth: (user: User, token: string, hasNFTAccess: boolean) => Promise<void>;
  setNFTStatus: (hasAccess: boolean, tokens?: string[]) => Promise<void>;
  logout: () => Promise<void>;
  refreshUser: () => Promise<void>;
}

export const useAuthStore = create<AuthState>((set, get) => ({
  user: null,
  token: null,
  hasNFTAccess: false,
  nftTokens: [],
  isAuthenticated: false,
  isLoading: true,

  setAuth: async (user: User, token:string, hasNFTAccess: boolean) => {
    await updateAuthToken(token);
    await SecureStore.setItemAsync("authToken", token);
    await SecureStore.setItemAsync("userData", JSON.stringify(user));
    await SecureStore.setItemAsync(
      "hasNFTAccess",
      JSON.stringify(hasNFTAccess)
    );
    // Reset NFT tokens on new login
    await SecureStore.setItemAsync("nftTokens", JSON.stringify([]));

    set({
      user,
      token,
      hasNFTAccess,
      nftTokens: [],
      isAuthenticated: true,
      isLoading: false,
    });
  },

  setNFTStatus: async (hasNFTAccess: boolean, tokens: string[] = []) => {
    await SecureStore.setItemAsync(
      "hasNFTAccess",
      JSON.stringify(hasNFTAccess)
    );
    await SecureStore.setItemAsync("nftTokens", JSON.stringify(tokens));

    set({
      hasNFTAccess,
      nftTokens: tokens,
    });
  },

  logout: async () => {
    await updateAuthToken(null);
    await SecureStore.deleteItemAsync("authToken");
    await SecureStore.deleteItemAsync("userData");
    await SecureStore.deleteItemAsync("hasNFTAccess");
    await SecureStore.deleteItemAsync("nftTokens");
    set({
      user: null,
      token: null,
      hasNFTAccess: false,
      nftTokens: [],
      isAuthenticated: false,
      isLoading: false,
    });
  },

  refreshUser: async () => {
    const token = await SecureStore.getItemAsync("authToken");
    if (!token) {
      set({ isAuthenticated: false, isLoading: false });
      return;
    }

    try {
      await updateAuthToken(token);
      const { data } = await apolloClient.query({
        query: GET_ME,
        fetchPolicy: "network-only",
      });

      if (data?.me) {
        await SecureStore.setItemAsync("userData", JSON.stringify(data.me));
        set((state) => ({
          user: data.me,
          isAuthenticated: true,
          isLoading: false,
          hasNFTAccess: state.hasNFTAccess,
          nftTokens: state.nftTokens,
        }));
      } else {
        await get().logout();
      }
    } catch (error) {
      console.error("Failed to refresh user:", error);
      await get().logout();
    }
  },
}));

export const initializeAuth = async () => {
  try {
    const [token, userData, nftAccessData, tokensData] = await Promise.all([
      SecureStore.getItemAsync("authToken"),
      SecureStore.getItemAsync("userData"),
      SecureStore.getItemAsync("hasNFTAccess"),
      SecureStore.getItemAsync("nftTokens"),
    ]);

    const hasNFTAccess = nftAccessData ? JSON.parse(nftAccessData) : false;
    const nftTokens = tokensData ? JSON.parse(tokensData) : [];

    if (token && userData) {
      const user = JSON.parse(userData);
      useAuthStore.setState({
        user,
        token,
        hasNFTAccess,
        nftTokens,
        isAuthenticated: true,
        isLoading: false,
      });
      await useAuthStore.getState().refreshUser();
    } else {
      useAuthStore.setState({
        user: null,
        token: null,
        hasNFTAccess: false,
        nftTokens: [],
        isAuthenticated: false,
        isLoading: false,
      });
    }
  } catch (error) {
    console.error("Failed to initialize auth:", error);
    useAuthStore.setState({
      user: null,
      token: null,
      hasNFTAccess: false,
      nftTokens: [],
      isAuthenticated: false,
      isLoading: false,
    });
  }
};
